cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED ENV{AUMANET_HALOW_ROOT})
    message(FATAL_ERROR "Run tools/build.sh: AUMANET_HALOW_ROOT must point to the pinned esp-halow checkout")
endif()
set(EXTRA_COMPONENT_DIRS "$ENV{AUMANET_HALOW_ROOT}/halow")
if(DEFINED ENV{AUMANET_GATEWAY_COMPONENTS})
    # Explicit experimental profile only. Entire image shares the extra pbuf ABI.
    list(APPEND EXTRA_COMPONENT_DIRS "$ENV{AUMANET_GATEWAY_COMPONENTS}/lwip" "$ENV{AUMANET_GATEWAY_COMPONENTS}/esp_netif" "${CMAKE_CURRENT_LIST_DIR}/gateway")
    add_compile_options(-include "${CMAKE_CURRENT_LIST_DIR}/gateway/include/aumanet_gateway_build.h")
    include_directories("${CMAKE_CURRENT_LIST_DIR}/gateway/include")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(DEFINED ENV{AUMANET_GATEWAY_COMPONENTS})
    set(PROJECT_VER "0.6.0-gateway.15")
    if(DEFINED ENV{AUMANET_RELEASE_VERSION})
        set(PROJECT_VER "$ENV{AUMANET_RELEASE_VERSION}")
    endif()
    project(aumanet_bench VERSION 0.6.0)
else()
    set(PROJECT_VER "0.5.0")
    project(aumanet_bench VERSION 0.5.0)
endif()
